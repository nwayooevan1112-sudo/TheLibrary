        allGenres: "All Genres",
        bookCatalogTitle: "E-Book Bookshelf",

        noBooksFound: "No books available yet",
        noBooksDesc: "Access the Owner Panel to upload new PDF E-books.",
        articlesTitle: "Articles & Publications",
        noArticlesFound: "No articles published yet",
        aboutText: "The Library is a dedicated digital sanctuary for acquiring PDF E-books, historical resources, and insightful literary content.",
        adminLockTitle: "Owner Access Control",

    let currentLang = 'my';
    let books = JSON.parse(localStorage.getItem('the_library_books')) || [];
    let articles = JSON.parse(localStorage.getItem('the_library_articles')) || [];
    let isAdminLoggedIn = false;

    // Toggle Language
    function toggleLanguage() {
      currentLang = currentLang === 'my' ? 'en' : 'my';
      document.getElementById('lang-btn-text').innerText = currentLang === 'my' ? 'EN' : 'မြန်မာ';
      updateLanguageUI();
    }

    function updateLanguageUI() {
      document.querySelectorAll('[data-i18n]').forEach(el => {
        const key = el.getAttribute('data-i18n');
        if (translations[currentLang][key]) {
          el.innerText = translations[currentLang][key];
        }
      });

      document.querySelectorAll('[data-i18n-placeholder]').forEach(el => {
        const key = el.getAttribute('data-i18n-placeholder');
        if (translations[currentLang][key]) {
          el.placeholder = translations[currentLang][key];
        }
      });

      renderBooks();
      renderArticles();
    }

    // Switch Nav Tabs
    function switchTab(tabName) {
      document.getElementById('tab-catalog').classList.add('hidden');
      document.getElementById('tab-articles').classList.add('hidden');
      document.getElementById('tab-about').classList.add('hidden');

      document.getElementById('nav-catalog').className = "text-slate-400 hover:text-amber-400 flex items-center gap-2 px-3 py-2 rounded-lg transition";
      document.getElementById('nav-articles').className = "text-slate-400 hover:text-amber-400 flex items-center gap-2 px-3 py-2 rounded-lg transition";
      document.getElementById('nav-about').className = "text-slate-400 hover:text-amber-400 flex items-center gap-2 px-3 py-2 rounded-lg transition";

      document.getElementById(`tab-${tabName}`).classList.remove('hidden');
      document.getElementById(`nav-${tabName}`).className = "text-amber-400 flex items-center gap-2 px-3 py-2 rounded-lg bg-amber-500/10";
    }

    // Render Books
    function renderBooks() {
      const grid = document.getElementById('bookGrid');
      const emptyState = document.getElementById('emptyState');
      const searchVal = document.getElementById('searchInput').value.toLowerCase();
      const genreVal = document.getElementById('genreFilter').value;

      const filtered = books.filter(b => {
        const matchSearch = b.title.toLowerCase().includes(searchVal) || b.author.toLowerCase().includes(searchVal);
        const matchGenre = genreVal === 'All' || b.genre === genreVal;
        return matchSearch && matchGenre;
      });

      document.getElementById('bookCount').innerText = filtered.length;

      if (filtered.length === 0) {
        grid.innerHTML = '';
        emptyState.classList.remove('hidden');
        return;
      }

      emptyState.classList.add('hidden');
      grid.innerHTML = filtered.map((book) => `
        <div class="bg-slate-900 rounded-xl overflow-hidden border border-slate-800 hover:border-amber-500/50 transition duration-300 flex flex-col group">
          <div class="h-48 bg-slate-950 relative overflow-hidden flex items-center justify-center">
            ${book.cover ? `<img src="${book.cover}" class="w-full h-full object-cover group-hover:scale-105 transition duration-500">` : `<i class="fa-solid fa-book text-4xl text-slate-700"></i>`}
            <span class="absolute top-2 right-2 bg-slate-950/80 backdrop-blur-md border border-slate-800 text-amber-400 text-[10px] px-2 py-0.5 rounded font-medium">
              ${book.genre}
            </span>
          </div>
          <div class="p-4 flex-1 flex flex-col justify-between space-y-3">
            <div>
              <h3 class="font-bold text-slate-200 text-sm line-clamp-1 group-hover:text-amber-400 transition">${book.title}</h3>
              <p class="text-xs text-slate-400 mt-1"><i class="fa-solid fa-pen-nib text-amber-500/70 mr-1"></i> ${book.author}</p>
              <p class="text-xs text-slate-500 mt-2 line-clamp-2">${book.desc || ''}</p>
            </div>
            <div class="pt-2 border-t border-slate-800/80 flex items-center justify-between gap-2">
              <a href="${book.pdfUrl}" target="_blank" class="flex-1 bg-amber-500 hover:bg-amber-600 text-slate-950 font-bold py-1.5 px-3 rounded-lg text-xs text-center transition flex items-center justify-center gap-1.5">
                <i class="fa-solid fa-download"></i> ${translations[currentLang].downloadBtn}
              </a>
              ${isAdminLoggedIn ? `
                <button onclick="deleteBook(${book.id})" class="p-1.5 text-rose-400 hover:bg-rose-500/10 rounded-lg transition" title="Delete">
                  <i class="fa-solid fa-trash"></i>
                </button>
              ` : ''}
            </div>
          </div>
        </div>
      `).join('');
    }

    // Render Articles
    function renderArticles() {
      const grid = document.getElementById('articlesGrid');
      const emptyArticles = document.getElementById('emptyArticles');

      if (articles.length === 0) {
        grid.innerHTML = '';
        emptyArticles.classList.remove('hidden');
        return;
      }

      emptyArticles.classList.add('hidden');
      grid.innerHTML = articles.map((art) => `
        <div class="bg-slate-900 p-5 rounded-xl border border-slate-800 space-y-3 relative">
          <div class="flex items-center justify-between text-xs text-slate-500">
            <span><i class="fa-solid fa-calendar text-sky-400 mr-1"></i> ${art.date}</span>
            ${isAdminLoggedIn ? `<button onclick="deleteArticle(${art.id})" class="text-rose-400 hover:underline text-xs"><i class="fa-solid fa-trash"></i> ${translations[currentLang].deleteBtn}</button>` : ''}
          </div>
          <h3 class="font-bold text-slate-200 text-base">${art.title}</h3>
          <p class="text-xs text-slate-400 leading-relaxed whitespace-pre-line">${art.content}</p>
        </div>
      `).join('');
    }

    // Admin Access Modal Control
    function openAdminModal() {
      document.getElementById('adminModal').classList.remove('hidden');
      if (isAdminLoggedIn) {
        document.getElementById('adminAuthView').classList.add('hidden');
        document.getElementById('adminDashboardView').classList.remove('hidden');
      } else {
        document.getElementById('adminAuthView').classList.remove('hidden');
        document.getElementById('adminDashboardView').classList.add('hidden');
      }
    }

    function closeAdminModal() {
      document.getElementById('adminModal').classList.add('hidden');
    }

    function verifyAdmin() {
      const passcode = document.getElementById('adminPasscode').value;
      if (passcode === 'admin123') {
        isAdminLoggedIn = true;
        document.getElementById('adminAuthView').classList.add('hidden');
        document.getElementById('adminDashboardView').classList.remove('hidden');
        renderBooks();
        renderArticles();
      } else {
        alert('Passcode မှားယွင်းနေပါသည်။');
      }
    }

    function logoutAdmin() {
      isAdminLoggedIn = false;
      closeAdminModal();
      renderBooks();
      renderArticles();
    }

    // Add Book
    function handleAddBook(e) {
      e.preventDefault();
      const newBook = {
        id: Date.now(),
        title: document.getElementById('bookTitle').value,
        author: document.getElementById('bookAuthor').value,
        genre: document.getElementById('bookGenre').value,
        cover: document.getElementById('bookCover').value || '',
        pdfUrl: document.getElementById('bookPdf').value,
        desc: document.getElementById('bookDesc').value
      };

      books.unshift(newBook);
      localStorage.setItem('the_library_books', JSON.stringify(books));
      renderBooks();
      e.target.reset();
      alert('E-Book တင်ပြီးပါပြီ။');
    }

    // Add Article
    function handleAddArticle(e) {
      e.preventDefault();
      const newArticle = {
        id: Date.now(),
        title: document.getElementById('articleTitle').value,
        content: document.getElementById('articleContent').value,
        date: new Date().toLocaleDateString()
      };

      articles.unshift(newArticle);
      localStorage.setItem('the_library_articles', JSON.stringify(articles));
      renderArticles();
      e.target.reset();
      alert('ဆောင်းပါး တင်ပြီးပါပြီ။');
    }

    // Delete Handlers
    function deleteBook(id) {
      if (confirm('ဤစာအုပ်ကို ဖျက်ရန် သေချာပါသလား။')) {
        books = books.filter(b => b.id !== id);
        localStorage.setItem('the_library_books', JSON.stringify(books));
        renderBooks();
      }
    }

    function deleteArticle(id) {
      if (confirm('ဤဆောင်းပါးကို ဖျက်ရန် သေချာပါသလား။')) {
        articles = articles.filter(a => a.id !== id);
        localStorage.setItem('the_library_articles', JSON.stringify(articles));
        renderArticles();
      }
    }

    // Search and Filter Handlers
    function handleSearch() { renderBooks(); }
    function handleFilter() { renderBooks(); }

    // Initial Load
    renderBooks();
    renderArticles();
